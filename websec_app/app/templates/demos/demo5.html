ℹ️Cómo funciona Fernet
from cryptography.fernet import Fernet # Fernet = AES-128-CBC + HMAC-SHA256 # Genera clave de 32 bytes (256 bits) key = Fernet.generate_key() f = Fernet(key) # Cifrar — el token incluye IV aleatorio + HMAC token = f.encrypt(b"dato sensible") # Descifrar — verifica HMAC automáticamente # Si el token fue modificado → InvalidToken plaintext = f.decrypt(token) # lanza excepción si corrupto
